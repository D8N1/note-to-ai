impl SignalClient {
    pub async fn connect(&self) -> Result<()> {
        // REAL implementation - no stubs
        let output = tokio::process::Command::new("signal-cli")
            .args(&["--config", &self.config_dir.to_str().unwrap(), "daemon", "--receive-mode", "on-connection"])
            .stdout(Stdio::piped())
            .stderr(Stdio::piped())
            .spawn()?;
            
        // Wait for actual connection
        let status = output.wait().await?;
        if !status.success() {
            return Err(SignalError::ConnectionFailed("Signal daemon failed to start".to_string()));
        }
        
        // Verify connection by sending test message
        self.verify_connection().await?;
        Ok(())
    }
    
    pub async fn receive_messages(&self) -> Result<Vec<SignalMessage>> {
        // REAL JSON parsing of Signal messages
        let output = tokio::process::Command::new("signal-cli")
            .args(&["--config", &self.config_dir.to_str().unwrap(), "receive", "--json"])
            .output()
            .await?;
            
        let stdout = String::from_utf8(output.stdout)?;
        let mut messages = Vec::new();
        
        for line in stdout.lines() {
            if line.trim().is_empty() { continue; }
            let json: serde_json::Value = serde_json::from_str(line)?;
            
            if let Some(envelope) = json.get("envelope") {
                messages.push(self.parse_signal_envelope(envelope)?);
            }
        }
        
        Ok(messages)
    }
    
    pub async fn send_message(&self, recipient: &str, message: &str) -> Result<()> {
        // REAL message sending
        let output = tokio::process::Command::new("signal-cli")
            .args(&["--config", &self.config_dir.to_str().unwrap(), "send", "-m", message, recipient])
            .output()
            .await?;
            
        if !output.status.success() {
            let stderr = String::from_utf8_lossy(&output.stderr);
            return Err(SignalError::SendFailed(stderr.to_string()));
        }
        
        Ok(())
    }
    
    // ADD: Real group chat support
    pub async fn create_group(&self, name: &str, members: &[&str]) -> Result<String> {
        let members_args: Vec<String> = members.iter().map(|m| m.to_string()).collect();
        let mut args = vec!["--config".to_string(), self.config_dir.to_str().unwrap().to_string(), "updateGroup".to_string(), "-n".to_string(), name.to_string()];
        for member in members_args {
            args.push("-m".to_string());
            args.push(member);
        }
        
        let output = tokio::process::Command::new("signal-cli")
            .args(&args)
            .output()
            .await?;
            
        if !output.status.success() {
            let stderr = String::from_utf8_lossy(&output.stderr);
            return Err(SignalError::GroupCreationFailed(stderr.to_string()));
        }
        
        // Parse group ID from output
        let stdout = String::from_utf8(output.stdout)?;
        self.extract_group_id(&stdout)
    }

    pub async fn transcribe_audio(&self, audio_path: &std::path::Path) -> Result<String> {
        // REAL Whisper transcription - no mocks
        let whisper = self.whisper_model.as_ref()
            .ok_or(AIError::ModelNotLoaded("Whisper"))?;
            
        // Load and preprocess audio
        let audio_data = self.load_audio_file(audio_path).await?;
        let mel_spectrogram = self.compute_mel_spectrogram(&audio_data)?;
        
        // Run actual inference
        let tokens = whisper.forward(&mel_spectrogram)?;
        let transcription = self.decode_tokens(&tokens)?;
        
        Ok(transcription)
    }
    
    pub async fn generate_embeddings(&self, text: &str) -> Result<Vec<f32>> {
        // REAL embedding generation - no mocks
        let model = self.embedding_model.as_ref()
            .ok_or(AIError::ModelNotLoaded("Embeddings"))?;
            
        // Tokenize text
        let tokens = self.tokenize_text(text)?;
        
        // Run inference
        let embeddings = model.forward(&tokens)?;
        
        // Convert to Vec<f32>
        let embedding_vec = embeddings.to_vec1::<f32>()?;
        
        Ok(embedding_vec)
    }

        let model = whisper::Model::load(&model_data, device)?;
        
        Ok(model)

use duckdb::{Connection, Result as DuckResult};
use lance::dataset::{Dataset, WriteMode};
use std::sync::Arc;

pub struct HybridStorageEngine {
    // REAL database connections - no phantoms
    duck_conn: Arc<Connection>,
    lance_dataset: Option<Dataset>,
    data_dir: std::path::PathBuf,
}

impl HybridStorageEngine {
    pub async fn new(data_dir: std::path::PathBuf) -> Result<Self> {
        // REAL database initialization
        std::fs::create_dir_all(&data_dir)?;
        
        // Initialize DuckDB for structured data
        let duck_path = data_dir.join("structured.duckdb");
        let duck_conn = Connection::open(&duck_path)?;
        
        // Create tables for medical data
        duck_conn.execute_batch(r#"
            CREATE TABLE IF NOT EXISTS conversations (
                id UUID PRIMARY KEY,
                group_id VARCHAR,
                sender VARCHAR,
                content TEXT,
                timestamp TIMESTAMP,
                medical_context JSONB
            );
            
            CREATE TABLE IF NOT EXISTS medical_entities (
                id UUID PRIMARY KEY,
                conversation_id UUID,
                entity_type VARCHAR,
                entity_value VARCHAR,
                confidence FLOAT,
                FOREIGN KEY (conversation_id) REFERENCES conversations(id)
            );
            
            CREATE TABLE IF NOT EXISTS audit_log (
                id UUID PRIMARY KEY,
                action VARCHAR,
                user_id VARCHAR,
                timestamp TIMESTAMP,
                details JSONB
            );
        "#)?;
        
        // Initialize Lance for vector storage
        let lance_path = data_dir.join("vectors.lance");
        let lance_dataset = if lance_path.exists() {
            Some(Dataset::open(&lance_path).await?)
        } else {
            None // Will be created on first write
        };
        
        Ok(Self {
            duck_conn: Arc::new(duck_conn),
            lance_dataset,
            data_dir,
        })
    }
    
    pub async fn store_conversation(&self, conversation: &Conversation) -> Result<()> {
        // REAL database insert
        let conn = &*self.duck_conn;
        
        conn.execute(
            "INSERT INTO conversations (id, group_id, sender, content, timestamp, medical_context) VALUES (?, ?, ?, ?, ?, ?)",
            &[
                &conversation.id.to_string(),
                &conversation.group_id,
                &conversation.sender,
                &conversation.content,
                &conversation.timestamp.to_rfc3339(),
                &serde_json::to_string(&conversation.medical_context)?,
            ],
        )?;
        
        Ok(())
    }
    
    pub async fn store_embeddings(&mut self, embeddings: &[EmbeddingRecord]) -> Result<()> {
        // REAL vector storage in Lance
        use arrow::array::{StringArray, Float32Array};
        use arrow::record_batch::RecordBatch;
        
        let ids: StringArray = embeddings.iter().map(|e| e.id.as_str()).collect();
        let vectors: Vec<Vec<f32>> = embeddings.iter().map(|e| e.vector.clone()).collect();
        let flat_vectors: Vec<f32> = vectors.into_iter().flatten().collect();
        let vector_array = Float32Array::from(flat_vectors);
        
        let batch = RecordBatch::try_new(
            Arc::new(Schema::new(vec![
                Field::new("id", DataType::Utf8, false),
                Field::new("vector", DataType::List(Box::new(Field::new("item", DataType::Float32, true))), false),
            ])),
            vec![Arc::new(ids), Arc::new(vector_array)],
        )?;
        
        if self.lance_dataset.is_none() {
            let lance_path = self.data_dir.join("vectors.lance");
            self.lance_dataset = Some(
                Dataset::write(&[batch], &lance_path, Some(WriteMode::Create)).await?
            );
        } else {
            self.lance_dataset.as_mut().unwrap().append(&[batch]).await?;
        }
        
        Ok(())
    }
    
    pub async fn search_conversations(&self, query: &str, limit: usize) -> Result<Vec<Conversation>> {
        // REAL database query
        let conn = &*self.duck_conn;
        
        let mut stmt = conn.prepare(
            "SELECT id, group_id, sender, content, timestamp, medical_context 
             FROM conversations 
             WHERE content LIKE ? 
             ORDER BY timestamp DESC 
             LIMIT ?"
        )?;
        
        let rows = stmt.query_map(&[&format!("%{}%", query), &limit.to_string()], |row| {
            Ok(Conversation {
                id: uuid::Uuid::parse_str(&row.get::<_, String>(0)?)?,
                group_id: row.get(1)?,
                sender: row.get(2)?,
                content: row.get(3)?,
                timestamp: chrono::DateTime::parse_from_rfc3339(&row.get::<_, String>(4)?)?.with_timezone(&chrono::Utc),
                medical_context: serde_json::from_str(&row.get::<_, String>(5)?)?,
            })
        })?;
        
        let mut conversations = Vec::new();
        for row in rows {
            conversations.push(row?);
        }
        
        Ok(conversations)
    }
}

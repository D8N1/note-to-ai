use std::sync::Arc;
use serde::{Deserialize, Serialize};

// Keep Arkworks for core cryptographic infrastructure
use ark_bls12_381::{Bls12_381, Fr};
use ark_groth16::{Groth16, Proof, ProvingKey, VerifyingKey};
use ark_crypto_primitives::snark::SNARK;

// Add Barretenberg for user-facing ZK proving
use noirc_abi::{Abi, InputMap};
use noirc_driver::CompileOptions;

#[derive(Debug, Clone)]
pub enum ProvingBackend {
    Arkworks(ArkworksBackend),
    Barretenberg(BarretenbergBackend),
}

#[derive(Debug, Clone)]
pub struct ArkworksBackend {
    proving_key: Arc<ProvingKey<Bls12_381>>,
    verifying_key: Arc<VerifyingKey<Bls12_381>>,
}

#[derive(Debug, Clone)]
pub struct BarretenbergBackend {
    circuit_bytecode: Vec<u8>,
    abi: Abi,
    backend_type: NoirBackendType,
}

#[derive(Debug, Clone)]
pub enum NoirBackendType {
    UltraHonk,
    UltraPlonk,
}

// Integration trait for seamless backend switching
pub trait ZkProver {
    type Proof: Serialize + for<'de> Deserialize<'de>;
    type PublicInputs: Serialize + for<'de> Deserialize<'de>;
    type PrivateInputs: Serialize + for<'de> Deserialize<'de>;
    
    async fn prove(
        &self,
        public_inputs: Self::PublicInputs,
        private_inputs: Self::PrivateInputs,
    ) -> Result<Self::Proof, ProofError>;
    
    async fn verify(
        &self,
        proof: &Self::Proof,
        public_inputs: &Self::PublicInputs,
    ) -> Result<bool, ProofError>;
    
    fn backend_type(&self) -> &str;
    fn is_mobile_optimized(&self) -> bool;
}

impl ZkProver for ProvingBackend {
    type Proof = UnifiedProof;
    type PublicInputs = serde_json::Value;
    type PrivateInputs = serde_json::Value;
    
    async fn prove(
        &self,
        public_inputs: Self::PublicInputs,
        private_inputs: Self::PrivateInputs,
    ) -> Result<Self::Proof, ProofError> {
        match self {
            ProvingBackend::Arkworks(backend) => {
                backend.prove_arkworks(public_inputs, private_inputs).await
            }
            ProvingBackend::Barretenberg(backend) => {
                backend.prove_noir(public_inputs, private_inputs).await
            }
        }
    }
    
    async fn verify(
        &self,
        proof: &Self::Proof,
        public_inputs: &Self::PublicInputs,
    ) -> Result<bool, ProofError> {
        match (self, proof) {
            (ProvingBackend::Arkworks(backend), UnifiedProof::Arkworks(proof)) => {
                backend.verify_arkworks(proof, public_inputs).await
            }
            (ProvingBackend::Barretenberg(backend), UnifiedProof::Barretenberg(proof)) => {
                backend.verify_noir(proof, public_inputs).await
            }
            _ => Err(ProofError::BackendMismatch),
        }
    }
    
    fn backend_type(&self) -> &str {
        match self {
            ProvingBackend::Arkworks(_) => "arkworks-groth16",
            ProvingBackend::Barretenberg(b) => match b.backend_type {
                NoirBackendType::UltraHonk => "barretenberg-ultrahonk",
                NoirBackendType::UltraPlonk => "barretenberg-ultraplonk",
            }
        }
    }
    
    fn is_mobile_optimized(&self) -> bool {
        match self {
            ProvingBackend::Arkworks(_) => false, // Heavy for mobile
            ProvingBackend::Barretenberg(_) => true, // Optimized for mobile/browser
        }
    }
}

#[derive(Debug, Serialize, Deserialize)]
pub enum UnifiedProof {
    Arkworks(Vec<u8>), // Serialized Arkworks proof
    Barretenberg(Vec<u8>), // Serialized Barretenberg proof
}

// Backend selection strategy
pub struct ProofStrategy {
    device_capabilities: DeviceCapabilities,
    use_case: ProofUseCase,
}

#[derive(Debug, Clone)]
pub struct DeviceCapabilities {
    available_memory_gb: f32,
    is_mobile: bool,
    supports_wasm: bool,
    supports_multithreading: bool,
}

#[derive(Debug, Clone)]
pub enum ProofUseCase {
    CoreInfrastructure,  // Use Arkworks for robustness
    UserFacingProof,     // Use Barretenberg for UX
    MobileProof,         // Always use Barretenberg
    HighSecurityProof,   // Use Arkworks for maximum security
}

impl ProofStrategy {
    pub fn select_backend(&self) -> ProvingBackend {
        match (&self.use_case, &self.device_capabilities) {
            // Mobile devices: Always use Barretenberg
            (_, caps) if caps.is_mobile => {
                ProvingBackend::Barretenberg(BarretenbergBackend::default_honk())
            }
            
            // User-facing proofs: Prefer Barretenberg for UX
            (ProofUseCase::UserFacingProof, caps) if caps.available_memory_gb < 8.0 => {
                ProvingBackend::Barretenberg(BarretenbergBackend::default_honk())
            }
            
            // Core infrastructure: Use Arkworks for robustness
            (ProofUseCase::CoreInfrastructure, _) => {
                ProvingBackend::Arkworks(ArkworksBackend::default())
            }
            
            // High security: Use Arkworks
            (ProofUseCase::HighSecurityProof, _) => {
                ProvingBackend::Arkworks(ArkworksBackend::default())
            }
            
            // Default: Barretenberg for better UX
            _ => ProvingBackend::Barretenberg(BarretenbergBackend::default_honk()),
        }
    }
}

// Implementation for existing zkPassport integration
impl BarretenbergBackend {
    pub fn default_honk() -> Self {
        Self {
            circuit_bytecode: Vec::new(), // Load from compiled Noir circuit
            abi: Abi::default(),
            backend_type: NoirBackendType::UltraHonk,
        }
    }
    
    async fn prove_noir(
        &self,
        public_inputs: serde_json::Value,
        private_inputs: serde_json::Value,
    ) -> Result<UnifiedProof, ProofError> {
        // Integration with Noir/Barretenberg
        use bb_rs::{Barretenberg, BackendOptions};
        
        let backend = Barretenberg::new(BackendOptions {
            backend_type: bb_rs::BackendType::UltraHonk,
            threads: if cfg!(target_os = "ios") { 1 } else { num_cpus::get() },
        })?;
        
        // Convert inputs to Noir format
        let input_map = self.serialize_inputs(public_inputs, private_inputs)?;
        
        // Generate proof using Barretenberg UltraHonk
        let proof = backend.prove(&self.circuit_bytecode, &input_map).await?;
        
        Ok(UnifiedProof::Barretenberg(proof))
    }
    
    fn serialize_inputs(
        &self,
        public_inputs: serde_json::Value,
        private_inputs: serde_json::Value,
    ) -> Result<InputMap, ProofError> {
        // Convert your existing input format to Noir's InputMap
        let mut input_map = InputMap::new();
        
        // Map your zkPassport inputs to Noir format
        if let Some(passport_data) = private_inputs.get("passport_data") {
            input_map.insert("passport_data".to_string(), passport_data.clone().into());
        }
        
        if let Some(challenge) = public_inputs.get("challenge") {
            input_map.insert("challenge".to_string(), challenge.clone().into());
        }
        
        Ok(input_map)
    }
}

// Error handling
#[derive(Debug, thiserror::Error)]
pub enum ProofError {
    #[error("Backend mismatch between proof and verifier")]
    BackendMismatch,
    
    #[error("Arkworks error: {0}")]
    Arkworks(String),
    
    #[error("Barretenberg error: {0}")]
    Barretenberg(String),
    
    #[error("Serialization error: {0}")]
    Serialization(#[from] serde_json::Error),
    
    #[error("Out of memory")]
    OutOfMemory,
}

// Integration with your existing zkPassport.rs
pub struct ZkPassportProver {
    strategy: ProofStrategy,
    backend: ProvingBackend,
}

impl ZkPassportProver {
    pub fn new(device_capabilities: DeviceCapabilities) -> Self {
        let strategy = ProofStrategy {
            device_capabilities: device_capabilities.clone(),
            use_case: if device_capabilities.is_mobile {
                ProofUseCase::MobileProof
            } else {
                ProofUseCase::UserFacingProof
            },
        };
        
        let backend = strategy.select_backend();
        
        Self { strategy, backend }
    }
    
    pub async fn prove_passport_signature(
        &self,
        passport_data: PassportData,
        challenge: Challenge,
    ) -> Result<PassportProof, ProofError> {
        let public_inputs = serde_json::json!({
            "challenge": challenge.to_bytes(),
            "timestamp": chrono::Utc::now().timestamp()
        });
        
        let private_inputs = serde_json::json!({
            "passport_data": passport_data.to_bytes(),
            "signature": passport_data.signature,
            "merkle_path": passport_data.merkle_path
        });
        
        let proof = self.backend.prove(public_inputs, private_inputs).await?;
        
        Ok(PassportProof {
            proof,
            backend_type: self.backend.backend_type().to_string(),
            timestamp: chrono::Utc::now(),
        })
    }
}

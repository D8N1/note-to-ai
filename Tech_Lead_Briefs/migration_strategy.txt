// Input structure matching your existing PassportData
struct PassportInputs {
    // Public inputs
    challenge: [u8; 32],
    country_code: [u8; 3],
    min_age: u8,
    
    // Private inputs  
    passport_signature: [u8; 64],
    passport_pubkey: [u8; 64],
    date_of_birth: [u8; 8],
    document_hash: [u8; 32],
    merkle_path: [[u8; 32]; 8],
    merkle_indices: [u1; 8],
}

fn main(inputs: PassportInputs) -> pub [u8; 32] {
    // 1. Verify passport signature (using built-in ECDSA verification)
    let message_hash = std::hash::pedersen_hash([
        inputs.document_hash,
        inputs.challenge
    ]);
    
    // Use Noir's built-in ECDSA verification for P-256 (passport standard)
    let signature_valid = std::ecdsa_secp256r1::verify_signature(
        inputs.passport_pubkey,
        inputs.passport_signature,
        message_hash.to_be_bytes()
    );
    assert(signature_valid);
    
    // 2. Verify age without revealing birth date
    let current_year = 2025; // Could be passed as public input
    let birth_year = u16::from_be_bytes([inputs.date_of_birth[0], inputs.date_of_birth[1]]);
    let age = current_year - birth_year;
    assert(age >= inputs.min_age as u16);
    
    // 3. Verify country if needed (optional constraint)
    // assert(inputs.country_code == expected_country);
    
    // 4. Verify document is in valid passport certificate chain
    let computed_root = compute_merkle_root(
        inputs.document_hash,
        inputs.merkle_path,
        inputs.merkle_indices
    );
    
    // Return proof commitment
    std::hash::pedersen_hash([
        inputs.challenge,
        computed_root,
        [age as u8; 32] // Age proof without revealing exact age
    ]).to_be_bytes()
}

fn compute_merkle_root(
    leaf: [u8; 32],
    path: [[u8; 32]; 8],
    indices: [u1; 8]
) -> [u8; 32] {
    let mut current = leaf;
    for i in 0..8 {
        let path_element = path[i];
        if indices[i] == 0 {
            current = std::hash::pedersen_hash([current, path_element]).to_be_bytes();
        } else {
            current = std::hash::pedersen_hash([path_element, current]).to_be_bytes();
        }
    }
    current
}

// Migration wrapper for your existing zkPassport implementation
pub struct MigratedZkPassport {
    // Keep existing Arkworks backend for compatibility
    arkworks_backend: Option<ArkworksPassportProver>,
    
    // New Barretenberg backend for mobile optimization
    barretenberg_backend: Option<BarretenbergPassportProver>,
    
    // Selection strategy
    device_capabilities: DeviceCapabilities,
}

impl MigratedZkPassport {
    pub async fn new(device_capabilities: DeviceCapabilities) -> Result<Self, MigrationError> {
        let mut instance = Self {
            arkworks_backend: None,
            barretenberg_backend: None,
            device_capabilities: device_capabilities.clone(),
        };
        
        // Initialize both backends for graceful fallback
        if !device_capabilities.is_mobile {
            // Keep Arkworks for server-side/desktop high-security operations
            instance.arkworks_backend = Some(
                ArkworksPassportProver::load_existing(&instance.get_arkworks_keys_path()).await?
            );
        }
        
        // Always initialize Barretenberg for better UX
        instance.barretenberg_backend = Some(
            BarretenbergPassportProver::compile_circuit(PASSPORT_NOIR_CIRCUIT).await?
        );
        
        Ok(instance)
    }
    
    // Smart backend selection based on context
    fn select_optimal_backend(&self, context: &ProofContext) -> &dyn PassportProver {
        match context {
            // Mobile or resource-constrained: Always use Barretenberg
            ProofContext::Mobile | ProofContext::ResourceConstrained => {
                self.barretenberg_backend.as_ref().unwrap()
            }
            
            // High security server-side operations: Use Arkworks if available
            ProofContext::HighSecurity if self.arkworks_backend.is_some() => {
                self.arkworks_backend.as_ref().unwrap()
            }
            
            // Default: Use Barretenberg for better performance
            _ => self.barretenberg_backend.as_ref().unwrap(),
        }
    }
    
    pub async fn prove_passport_valid(
        &self,
        passport_data: &PassportData,
        challenge: &[u8; 32],
        context: ProofContext,
    ) -> Result<PassportProof, MigrationError> {
        let backend = self.select_optimal_backend(&context);
        
        let proof = backend.prove(passport_data, challenge).await
            .map_err(MigrationError::ProofGeneration)?;
        
        Ok(PassportProof {
            proof_data: proof,
            backend_used: backend.backend_name().to_string(),
            proof_context: context,
            timestamp: chrono::Utc::now(),
        })
    }
    
    // Migration helper: Convert existing Arkworks circuits to Noir
    pub async fn migrate_existing_circuits(&self) -> Result<(), MigrationError> {
        if let Some(arkworks) = &self.arkworks_backend {
            let circuit_params = arkworks.export_circuit_parameters().await?;
            
            // Generate equivalent Noir circuit from Arkworks R1CS
            let noir_circuit = self.convert_r1cs_to_noir(&circuit_params)?;
            
            // Compile and verify equivalence
            let barretenberg = BarretenbergPassportProver::compile_circuit(&noir_circuit).await?;
            
            // Test with known inputs to ensure equivalence
            self.verify_migration_correctness(&arkworks, &barretenberg).await?;
        }
        Ok(())
    }
    
    fn get_arkworks_keys_path(&self) -> PathBuf {
        // Path to your existing Arkworks proving/verifying keys
        PathBuf::from("keys/passport_arkworks.pk")
    }
}

// Context-aware proving for optimal backend selection
#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum ProofContext {
    Mobile,              // iPhone/Android app
    ResourceConstrained, // Low-memory devices
    HighSecurity,       // Server-side verification
    UserFacing,         // Interactive web app
    BatchProcessing,    // Multiple proofs
}

// Unified trait for both backends
pub trait PassportProver {
    async fn prove(
        &self,
        passport_data: &PassportData,
        challenge: &[u8; 32],
    ) -> Result<Vec<u8>, Box<dyn std::error::Error>>;
    
    async fn verify(
        &self,
        proof: &[u8],
        public_inputs: &PassportPublicInputs,
    ) -> Result<bool, Box<dyn std::error::Error>>;
    
    fn backend_name(&self) -> &str;
    fn estimated_proving_time(&self, device: &DeviceCapabilities) -> std::time::Duration;
    fn estimated_memory_usage(&self) -> usize;
}

// Barretenberg implementation optimized for mobile
pub struct BarretenbergPassportProver {
    circuit_bytecode: Vec<u8>,
    abi: noirc_abi::Abi,
}

impl PassportProver for BarretenbergPassportProver {
    async fn prove(
        &self,
        passport_data: &PassportData,
        challenge: &[u8; 32],
    ) -> Result<Vec<u8>, Box<dyn std::error::Error>> {
        use bb_rs::{Barretenberg, BackendOptions, BackendType};
        
        let backend = Barretenberg::new(BackendOptions {
            backend_type: BackendType::UltraHonk, // Fastest for mobile
            threads: if cfg!(target_os = "ios") { 1 } else { 
                std::cmp::min(num_cpus::get(), 4) // Limit threads on mobile
            },
        })?;
        
        // Convert your PassportData to Noir inputs
        let input_map = self.passport_data_to_noir_inputs(passport_data, challenge)?;
        
        // Generate proof - optimized for mobile memory constraints
        let proof = backend.prove(&self.circuit_bytecode, &input_map).await?;
        
        Ok(proof)
    }
    
    async fn verify(
        &self,
        proof: &[u8],
        public_inputs: &PassportPublicInputs,
    ) -> Result<bool, Box<dyn std::error::Error>> {
        use bb_rs::Barretenberg;
        
        let backend = Barretenberg::new_verifier()?;
        let verification_key = backend.get_verification_key(&self.circuit_bytecode).await?;
        
        let public_input_map = self.public_inputs_to_noir_format(public_inputs)?;
        
        Ok(backend.verify(proof, &verification_key, &public_input_map).await?)
    }
    
    fn backend_name(&self) -> &str {
        "barretenberg-ultrahonk"
    }
    
    fn estimated_proving_time(&self, device: &DeviceCapabilities) -> std::time::Duration {
        if device.is_mobile {
            if device.available_memory_gb < 4.0 {
                std::time::Duration::from_secs(15) // Low-end mobile
            } else {
                std::time::Duration::from_secs(8)  // Modern mobile
            }
        } else {
            std::time::Duration::from_secs(3) // Desktop
        }
    }
    
    fn estimated_memory_usage(&self) -> usize {
        1024 * 1024 * 512 // ~512MB peak memory usage
    }
}

impl BarretenbergPassportProver {
    pub async fn compile_circuit(noir_source: &str) -> Result<Self, MigrationError> {
        // Compile Noir circuit to bytecode
        let compiled = noirc_driver::compile_contracts(
            noir_source,
            &noirc_driver::CompileOptions::default()
        ).map_err(MigrationError::CircuitCompilation)?;
        
        Ok(Self {
            circuit_bytecode: compiled.bytecode,
            abi: compiled.abi,
        })
    }
    
    fn passport_data_to_noir_inputs(
        &self,
        passport_data: &PassportData,
        challenge: &[u8; 32],
    ) -> Result<noirc_abi::InputMap, MigrationError> {
        let mut input_map = noirc_abi::InputMap::new();
        
        // Map your existing PassportData fields to Noir circuit inputs
        input_map.insert("challenge".to_string(), 
            noirc_abi::InputValue::Array(
                challenge.iter().map(|&b| noirc_abi::InputValue::Field(b.into())).collect()
            )
        );
        
        input_map.insert("passport_signature".to_string(),
            noirc_abi::InputValue::Array(
                passport_data.signature.iter().map(|&b| noirc_abi::InputValue::Field(b.into())).collect()
            )
        );
        
        // Add other required fields...
        
        Ok(input_map)
    }
}

// src/signal/message_processor.rs - New file

use crate::signal::conversational_assistant::ConversationalAssistant;

pub struct MessageProcessor {
    assistant: ConversationalAssistant,
    last_message_time: Option<DateTime<Utc>>,
}

impl MessageProcessor {
    pub async fn process_incoming_message(&mut self, raw_message: &str) -> Result<String, ProcessorError> {
        // Add human-like timing
        if let Some(last_time) = self.last_message_time {
            let time_since_last = Utc::now().signed_duration_since(last_time);
            if time_since_last < chrono::Duration::seconds(2) {
                // Add slight delay to feel more natural
                tokio::time::sleep(std::time::Duration::from_millis(1500)).await;
            }
        }
        
        // Create Signal message from raw input
        let message = SignalMessage {
            content: self.detect_content_type(raw_message).await?,
            timestamp: Utc::now(),
            sender: "Note to Self".to_string(),
        };
        
        // Process with assistant
        let response = self.assistant.process_signal_message(message).await?;
        
        // Update timing
        self.last_message_time = Some(Utc::now());
        
        Ok(response.content)
    }
    
    async fn detect_content_type(&self, raw_message: &str) -> Result<MessageContent, ProcessorError> {
        // Detect if it's voice, text, or image
        if raw_message.starts_with("voice:") {
            // Handle voice message
            let audio_data = self.decode_voice_message(raw_message).await?;
            Ok(MessageContent::Voice(audio_data))
        } else if raw_message.starts_with("image:") {
            // Handle image message
            let image_data = self.decode_image_message(raw_message).await?;
            Ok(MessageContent::Image(image_data))
        } else {
            // Handle text message
            Ok(MessageContent::Text(raw_message.to_string()))
        }
    }
}

// src/signal/timing.rs - New file for natural conversation timing

pub struct ConversationTiming {
    typing_speed_wpm: u32,      // Simulate human typing speed
    thinking_delay_ms: u64,     // Brief pause before complex responses
    voice_processing_delay_ms: u64, // Time for "listening" to voice notes
}

impl ConversationTiming {
    pub fn executive_assistant() -> Self {
        Self {
            typing_speed_wpm: 80,           // Fast, professional typing
            thinking_delay_ms: 800,         // Brief pause for analysis
            voice_processing_delay_ms: 1200, // Time to "process" voice note
        }
    }
    
    pub fn calculate_response_delay(&self, response_length: usize) -> Duration {
        // Calculate natural typing time
        let words = response_length / 5; // Average word length
        let typing_time_ms = (words as u64 * 60 * 1000) / self.typing_speed_wpm as u64;
        
        // Add thinking delay for complex responses
        let total_delay = if response_length > 500 {
            typing_time_ms + self.thinking_delay_ms
        } else {
            typing_time_ms.min(2000) // Cap at 2 seconds for short responses
        };
        
        Duration::from_millis(total_delay.max(500).min(5000)) // 0.5-5 second range
    }
    
    pub async fn simulate_voice_processing(&self, audio_duration: Duration) {
        // Simulate time to "listen" to voice note
        let processing_time = audio_duration.as_millis() as u64 / 4; // 4x real-time processing
        let delay = processing_time.max(self.voice_processing_delay_ms).min(3000);
        
        tokio::time::sleep(Duration::from_millis(delay)).await;
    }
}
Smart Interruption Management
// src/signal/interruption_manager.rs

pub struct InterruptionManager {
    last_user_message: Option<DateTime<Utc>>,
    user_timezone: chrono_tz::Tz,
    interruption_preferences: InterruptionPreferences,
}

#[derive(Debug, Clone)]
pub struct InterruptionPreferences {
    pub quiet_hours: (u8, u8),           // (start_hour, end_hour) in user timezone
    pub max_proactive_per_day: u32,
    pub urgency_threshold: UrgencyLevel,
    pub batch_low_priority: bool,         // Batch non-urgent insights
}

impl InterruptionManager {
    pub fn should_send_proactive_message(&self, insight: &ProactiveInsight) -> InterruptDecision {
        // Check quiet hours
        if self.is_quiet_hours() {
            return match insight.urgency {
                UrgencyLevel::Critical => InterruptDecision::SendImmediately,
                _ => InterruptDecision::QueueForMorning,
            };
        }
        
        // Check if user is actively messaging
        if let Some(last_msg) = self.last_user_message {
            let time_since = Utc::now().signed_duration_since(last_msg);
            if time_since < chrono::Duration::minutes(5) {
                return InterruptDecision::WaitForConversationPause;
            }
        }
        
        // Check daily limit
        if self.get_today_proactive_count() >= self.interruption_preferences.max_proactive_per_day {
            return match insight.urgency {
                UrgencyLevel::Critical | UrgencyLevel::High => InterruptDecision::SendImmediately,
                _ => InterruptDecision::QueueForTomorrow,
            };
        }
        
        // Check urgency threshold
        match insight.urgency {
            urgency if urgency >= self.interruption_preferences.urgency_threshold => {
                InterruptDecision::SendImmediately
            }
            _ if self.interruption_preferences.batch_low_priority => {
                InterruptDecision::BatchWithOthers
            }
            _ => InterruptDecision::Skip,
        }
    }
    
    fn is_quiet_hours(&self) -> bool {
        let now = Utc::now().with_timezone(&self.user_timezone);
        let current_hour = now.hour() as u8;
        let (start, end) = self.interruption_preferences.quiet_hours;
        
        if start < end {
            current_hour >= start && current_hour < end
        } else {
            // Handles overnight quiet hours (e.g., 22:00 to 07:00)
            current_hour >= start || current_hour < end
        }
    }
}

#[derive(Debug)]
pub enum InterruptDecision {
    SendImmediately,
    WaitForConversationPause,
    QueueForMorning,
    QueueForTomorrow,
    BatchWithOthers,
    Skip,
}

impl SignalClient {
    // Add to your existing implementation
    pub async fn start_intelligent_monitoring(&mut self) -> Result<(), SignalError> {
        let mut assistant = ConversationalAssistant::new().await?;
        let timing = ConversationTiming::executive_assistant();
        let mut interruption_manager = InterruptionManager::new(self.user_timezone);
        
        // Start the main conversation loop
        loop {
            tokio::select! {
                // Handle incoming messages from user
                Some(message) = self.receive_note_to_self() => {
                    interruption_manager.user_activity_detected();
                    
                    // Add natural processing delay for voice messages
                    if let MessageContent::Voice(audio) = &message.content {
                        timing.simulate_voice_processing(audio.duration).await;
                    }
                    
                    // Process message
                    let response = assistant.process_signal_message(message).await?;
                    
                    // Add natural response timing
                    let delay = timing.calculate_response_delay(response.content.len());
                    tokio::time::sleep(delay).await;
                    
                    // Send response
                    self.send_to_note_to_self(&response.content).await?;
                }
                
                // Handle calendar events for proactive assistance
                Some(event) = self.calendar_watcher.next_event() => {
                    if let Some(response) = assistant.process_calendar_event(event).await? {
                        if interruption_manager.should_send_proactive_message(&response.into()) 
                           == InterruptDecision::SendImmediately {
                            self.send_to_note_to_self(&response.content).await?;
                        }
                    }
                }
                
                // Handle background research completion
                Some(research_result) = assistant.background_research_completed() => {
                    let formatted_result = assistant.format_research_result(research_result).await?;
                    self.send_to_note_to_self(&formatted_result).await?;
                }
                
                // Periodic proactive insights (every 2 hours during active hours)
                _ = tokio::time::sleep(Duration::from_hours(2)) => {
                    if let Some(insight) = assistant.generate_periodic_insight().await? {
                        match interruption_manager.should_send_proactive_message(&insight) {
                            InterruptDecision::SendImmediately => {
                                self.send_to_note_to_self(&insight.format_as_message()).await?;
                            }
                            InterruptDecision::QueueForMorning => {
                                self.queue_for_morning_briefing(insight).await?;
                            }
                            _ => {} // Skip or handle other cases
                        }
                    }
                }
            }
        }
    }
}

// src/signal/morning_briefing.rs

pub struct MorningBriefingSystem {
    queued_insights: Vec<ProactiveInsight>,
    pending_actions: Vec<ActionItem>,
    calendar_preview: Vec<CalendarEvent>,
}

impl MorningBriefingSystem {
    pub async fn generate_morning_brief(&self) -> Result<String, BriefingError> {
        let mut brief = String::new();
        
        brief.push_str("🌅 **Good morning! Here's your strategic brief:**\n\n");
        
        // Priority items from yesterday
        if !self.pending_actions.is_empty() {
            brief.push_str("**Carry-forward items:**\n");
            for action in &self.pending_actions {
                brief.push_str(&format!("□ {} ({})\n", action.description, action.priority));
            }
            brief.push('\n');
        }
        
        // Today's calendar with context
        if !self.calendar_preview.is_empty() {
            brief.push_str("**Today's key meetings:**\n");
            for event in &self.calendar_preview {
                let context = self.find_meeting_context(&event.title).await?;
                brief.push_str(&format!("• {} - {}\n", event.title, context));
            }
            brief.push('\n');
        }
        
        // Insights that were queued overnight
        if !self.queued_insights.is_empty() {
            brief.push_str("**Strategic insights:**\n");
            for insight in &self.queued_insights {
                brief.push_str(&format!("💡 {}\n", insight.insight));
            }
            brief.push('\n');
        }
        
        brief.push_str("Ready to tackle the day! 🚀");
        
        Ok(brief)
    }
}
